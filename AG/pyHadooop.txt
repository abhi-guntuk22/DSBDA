//Mapper

import sys

for line in sys.stdin:
    line = line.strip()              # remove newline
    words = line.split(',')          # split by comma

    for word in words:
        word = word.strip()          # clean spaces/newlines
        if word:                     # skip empty values
            print(f"{word}\t1")





//Reducer

import sys
import io

current_word = None
current_count = 0
word = None

for line in sys.stdin:
    word, count = line.split("\t", 1)
    count = int(count)

    if current_word == word:
        current_count += count
    else:
        if current_word:
            print('%s\t%s' % (current_word, current_count))
        current_count = count
        current_word = word

if current_word == word:
    print('%s\t%s' % (current_word, current_count))
